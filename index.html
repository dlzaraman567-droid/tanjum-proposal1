    <script>
        // Initialize Lucide icons
        lucide.createIcons();

        // Reasons list for Step 2 rotator
        const reasons = [
            "\"The way your eyes light up when you talk about things you love.\"",
            "\"How safe and happy I feel whenever I'm right next to you.\"",
            "\"Your kindness and the gentle way you care for everyone around you.\"",
            "\"Every laugh we share makes my entire world infinitely better.\""
        ];
        let currentReasonIdx = 0;
        setInterval(() => {
            currentReasonIdx = (currentReasonIdx + 1) % reasons.length;
            const reasonEl = document.getElementById('reason-text');
            if (reasonEl) {
                reasonEl.style.opacity = '0';
                setTimeout(() => {
                    reasonEl.innerText = reasons[currentReasonIdx];
                    reasonEl.style.opacity = '1';
                }, 300);
            }
        }, 4000);

        // Floating hearts generator
        function createFloatingHearts() {
            const container = document.getElementById('hearts-container');
            const heartCount = 15;
            for (let i = 0; i < heartCount; i++) {
                const heart = document.createElement('div');
                heart.className = 'floating-heart';
                heart.innerHTML = '<svg xmlns="http://www.w3.org/2000/svg" width="24" height="24" viewBox="0 0 24 24" fill="currentColor"><path d="M19 14c1.49-1.46 3-3.21 3-5.5A5.5 5.5 0 0 0 16.5 3c-1.76 0-3 .5-4.5 2-1.5-1.5-2.74-2-4.5-2A5.5 5.5 0 0 0 2 8.5c0 2.3 1.5 4.04 3 5.5l7 7Z"/></svg>';
                
                // Random positioning and timing
                const size = Math.floor(Math.random() * 20) + 12; // 12px to 32px
                heart.style.width = `${size}px`;
                heart.style.height = `${size}px`;
                heart.style.left = `${Math.random() * 100}%`;
                heart.style.animationDuration = `${Math.random() * 8 + 6}s`; // 6s to 14s
                heart.style.animationDelay = `${Math.random() * 5}s`;
                
                container.appendChild(heart);
            }
        }
        createFloatingHearts();

        // Step Navigation Handler
        function nextStep(stepNumber) {
            document.querySelectorAll('.step-screen').forEach(el => {
                el.classList.add('hidden');
                el.classList.remove('flex');
            });
            const target = document.getElementById(`step-${stepNumber}`);
            if (target) {
                target.classList.remove('hidden');
                target.classList.add('flex');
            }
            // Re-initialize icons for newly displayed elements
            lucide.createIcons();
        }

        // Playful "No" button behaviors
        const noPleas = [
            "Are you sure, Tanjum? 🥺",
            "Think about all the ice cream dates! 🍦",
            "My heart will be so sad... 💔",
            "You missed the 'Yes' button! Right over there 👉",
            "Pretty please with sprinkles on top? ✨",
            "Nope, that button is broken, try the other one! 😉"
        ];
        let pleaIndex = 0;

        function moveNoButton() {
            const noBtn = document.getElementById('no-btn');
            const pleaEl = document.getElementById('no-plea');
            
            // Random translation within bounds
            const randomX = (Math.random() - 0.5) * 160;
            const randomY = (Math.random() - 0.5) * 80;
            
            noBtn.style.transform = `translate(${randomX}px, ${randomY}px)`;
            
            // Show cute plea message
            pleaEl.innerText = noPleas[pleaIndex];
            pleaIndex = (pleaIndex + 1) % noPleas.length;
            
            // Slightly grow the Yes button for persuasion
            const yesBtn = document.getElementById('yes-btn');
            const currentScale = 1 + (pleaIndex * 0.05);
            yesBtn.style.transform = `scale(${currentScale})`;
        }

        function handleNoClick() {
            moveNoButton();
        }

        // Celebration Trigger on "Yes"
        function celebrateSuccess() {
            nextStep(5);
            
            // Trigger glorious confetti celebration
            const duration = 4 * 1000;
            const animationEnd = Date.now() + duration;
            const defaults = { startVelocity: 30, spread: 360, ticks: 60, zIndex: 1000 };

            function randomInRange(min, max) {
                return Math.random() * (max - min) + min;
            }

            const interval = setInterval(function() {
                const timeLeft = animationEnd - Date.now();

                if (timeLeft <= 0) {
                    return clearInterval(interval);
                }

                const particleCount = 50 * (timeLeft / duration);
                
                // Confetti bursts in rose gold, pink, and gold tones
                confetti(Object.assign({}, defaults, { 
                    particleCount, 
                    origin: { x: randomInRange(0.1, 0.3), y: Math.random() - 0.2 },
                    colors: ['#ce5d5d', '#f5d1d1', '#ffd700', '#ff9494']
                }));
                confetti(Object.assign({}, defaults, { 
                    particleCount, 
                    origin: { x: randomInRange(0.7, 0.9), y: Math.random() - 0.2 },
                    colors: ['#ce5d5d', '#f5d1d1', '#ffd700', '#ff9494']
                }));
            }, 250);
        }

        // Reset App flow
        function resetApp() {
            pleaIndex = 0;
            document.getElementById('no-plea').innerText = "";
            const yesBtn = document.getElementById('yes-btn');
            yesBtn.style.transform = 'scale(1)';
            const noBtn = document.getElementById('no-btn');
            noBtn.style.transform = 'translate(0, 0)';
            nextStep(1);
        }
    </script>
